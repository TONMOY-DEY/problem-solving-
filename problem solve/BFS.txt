#include <iostream>
using namespace std;

int main() {

    int n = 5;

    int graph[6][6] = {0};

    graph[1][2] = graph[2][1] = 1;
    graph[1][3] = graph[3][1] = 1;
    graph[2][4] = graph[4][2] = 1;
    graph[3][4] = graph[4][3] = 1;

    int visited[6] = {0};

    int components = 0;

    for (int start = 1; start <= n; start++) {

        if (visited[start] == 0) {

            components++;

            int queue[6];
            int front = 0;
            int rear = 0;

            queue[rear++] = start;
            visited[start] = 1;

            cout << "Component " << components << ": ";

            while (front < rear) {

                int current = queue[front++];

                cout << current << " ";

                for (int i = 1; i <= n; i++) {

                    if (graph[current][i] == 1 &&
                        visited[i] == 0) {

                        queue[rear++] = i;
                        visited[i] = 1;
                    }
                }
            }

            cout << endl;
        }
    }

    cout << "Number of Connected Components: "
         << components << endl;

    return 0;
}
